local function looksLikeHitbox(p)
if not(p and p.Parent)then return false end
local char=LocalPlayer.Character
if char and p:IsDescendantOf(char)then return false end
local ok,res=pcall(function()
if p.Size.Magnitude<1 then return false end
local named=string.find(string.lower(p.Name),"hitbox",1,true)~=nil
if named then return true end
local m=p:FindFirstAncestorOfClass("Model")
if m and m:FindFirstChildOfClass("Humanoid")then return false end
return(not p.CanCollide)and p.Transparency>=0.8 and p.Size.Magnitude<=250
end)
return ok and res
end
local function farEdge(part,rootPos)
local off=part.Position-rootPos
local d=off.Magnitude
if d<0.01 then
return math.max(part.Size.X,part.Size.Z)/2
end
local u=part.CFrame:VectorToObjectSpace(off.Unit)
local s=part.Size
return d+math.abs(u.X)*s.X/2+math.abs(u.Y)*s.Y/2+math.abs(u.Z)*s.Z/2
end
local function insideHitbox(part,pos,margin)
local ok,res=pcall(function()
local rel=part.CFrame:PointToObjectSpace(pos)
local h=part.Size/2
return math.abs(rel.X)<=h.X+margin and math.abs(rel.Y)<=h.Y+margin and math.abs(rel.Z)<=h.Z+margin
end)
return ok and res
end
local function skillReach(id)
local learned=HitboxReach[id]
if CONFIG.AutoHitbox and learned then
return math.max(learned+CONFIG.HitboxMargin,CONFIG.SkillRange)
end
return CONFIG.SkillRange
end
local function rangeFor(kind,id)
if kind=="Skill"then return skillReach(id)end
if kind=="Front Dash"then return CONFIG.FrontDashRange end
if kind=="Slide Dash"then return CONFIG.SlideDashRange end
return CONFIG.Range
end
local function scanHitboxes(now)
if next(skillWatch)==nil then
if#recentParts>0 then recentParts={}end
return
end
local i=1
while i<=#recentParts do
local rp=recentParts[i]
if rp.ok==nil and now>rp.t then
rp.ok=looksLikeHitbox(rp.part)
end
if rp.ok==false or(now-rp.t)>(HITBOX_WINDOW+0.3)or not rp.part.Parent then
table.remove(recentParts,i)
else
i+=1
end
end
for _,rp in ipairs(recentParts)do
if rp.ok and not rp.done then
local best,bestD
for _,w in pairs(skillWatch)do
if w.root.Parent and rp.t>=w.startedAt-0.25 and(rp.t-w.startedAt)<=HITBOX_WINDOW then
local d=(rp.part.Position-w.root.Position).Magnitude
if d<=HITBOX_SCAN_RADIUS and(not bestD or d<bestD)then
best,bestD=w,d
end
end
end
if best then
rp.done=true
table.insert(best.parts,rp.part)
local reach=math.clamp(farEdge(rp.part,best.root.Position),4,120)
HitboxReach[best.id]=math.max(HitboxReach[best.id]or 0,reach)
dbg("hitbox",best.id,rp.part.Name,"reach",math.floor(reach))
end
end
end
end

local CAM_BIND="AutoBlockCamlock"
local PITCH_FOLLOW_LOCKED=0.01
local CAM_MAX_PITCH=math.rad(85)

if camState then stopCamlock()end
return
end
local cam=Workspace.CurrentCamera
local myRoot=getMyRoot()
local root=currentTarget and currentTarget.root
if not(alive and holding and CONFIG.Camlock and cam and myRoot and root and root.Parent)then
stopCamlock()

end
if not camState then
local pitch0,yaw0=cam.CFrame:ToOrientation()
camState={
origType=cam.CameraType,
rootInCam=cam.CFrame:PointToObjectSpace(myRoot.Position),
lockPitch=pitch0,
curPitch=pitch0,
curYaw=yaw0,
lastRot=cam.CFrame.Rotation,
}
cam.CameraType=Enum.CameraType.Scriptable
end
local st=camState
local k=1-math.exp(-CONFIG.CamlockSpeed*dt)
local dir=root.Position-myRoot.Position
if Vector3.new(dir.X,0,dir.Z).Magnitude>0.05 then
local targetYaw=math.atan2(-dir.X,-dir.Z)
local diff=(targetYaw-st.curYaw+math.pi)%(math.pi*2)-math.pi
st.curYaw=st.curYaw+diff*k
end
if PITCH_FOLLOW_LOCKED>0 then
local camPos=myRoot.Position-st.lastRot*st.rootInCam
local d3=root.Position-camPos
local horiz=math.sqrt(d3.X*d3.X+d3.Z*d3.Z)
local aim=math.clamp(math.atan2(d3.Y,math.max(horiz,0.01)),-CAM_MAX_PITCH,CAM_MAX_PITCH)
st.curPitch=st.curPitch+(st.lockPitch+(aim-st.lockPitch)*PITCH_FOLLOW_LOCKED-st.curPitch)*k
end
local rot=CFrame.fromOrientation(st.curPitch,st.curYaw,0)
st.lastRot=rot
cam.CFrame=CFrame.new(myRoot.Position-rot*st.rootInCam)*rot
end)

return(now-info.stoppedAt)>=math.max(CONFIG.ReleaseOffset,0)

local hooked=setmetatable({},{__mode="k"})
local seen=setmetatable({},{__mode="k"})
local function readPos(t)return t.TimePosition end
local function readPlaying(t)return t.IsPlaying end
local function markAndHandle(model,track,fromPoll)
local okp,pos=pcall(readPos,track)
if not okp then pos=0 end
local now=os.clock()
local rec=seen[track]
if fromPoll then
if not rec then
seen[track]={t=now,pos=pos}
onTrack(model,track)
else
if pos+0.08<rec.pos then
rec.t=now
onTrack(model,track)
end
rec.pos=pos
end
else
if rec and now-rec.t<0.1 then
rec.pos=pos
return
end
seen[track]={t=now,pos=pos}
onTrack(model,track)
end
end

local animatorCache=setmetatable({},{__mode="k"})
local rootCache=setmetatable({},{__mode="k"})
local pollFrame=0
local partConn=nil
setPartListener=function(on)
if on and not partConn then
partConn=Workspace.DescendantAdded:Connect(function(d)
if d:IsA("BasePart")then
recentParts[#recentParts+1]={part=d,t=os.clock()}
if#recentParts>60 then
table.remove(recentParts,1)
end
end
end)
elseif(not on)and partConn then
partConn:Disconnect()
partConn=nil
end
end

table.insert(connections,RunService.Heartbeat:Connect(function()
if not(alive and CONFIG.Enabled)then return end
local now=os.clock()
selTick(now)
scanHitboxes(now)
for track,w in pairs(skillWatch)do
if not w.model.Parent or(now-w.startedAt)>(HITBOX_WINDOW+0.5)then
skillWatch[track]=nil
end
end
if next(skillWatch)==nil then setPartListener(false)end
for track,p in pairs(pending)do
local okq,playing=pcall(readPlaying,track)
if not okq then playing=false end
local window=(p.kind=="Skill")and SKILL_PENDING_WINDOW or PENDING_WINDOW
if not playing or not p.model.Parent or(now-p.startedAt)>window then
pending[track]=nil
else
local myRoot=getMyRoot()
local root=p.model:FindFirstChild("HumanoidRootPart")
if myRoot and root then
local dist=(myRoot.Position-root.Position).Magnitude
local hit=dist<=rangeFor(p.kind,p.id)
if(not hit)and p.kind=="Skill"and CONFIG.AutoHitbox and p.watch then
for _,part in ipairs(p.watch.parts)do
if part.Parent and insideHitbox(part,myRoot.Position,CONFIG.HitboxMargin)then
hit=true
break
end
end
end
if hit then
pending[track]=nil
startBlock(p.model,root,track,p.kind,p.id,dist)
end
end
end
end
pollFrame+=1
local meRoot=getMyRoot()
if meRoot then
local myPos=meRoot.Position
local myChar=LocalPlayer.Character
local selMode=(CONFIG.Mode=="Select")
for model in pairs(trackedModels)do
if model.Parent and model~=myChar and(not selMode or isSelected(model))then
local mRoot=rootCache[model]
if not(mRoot and mRoot.Parent)then
mRoot=model:FindFirstChild("HumanoidRootPart")
rootCache[model]=mRoot
end
if mRoot then
local d=(mRoot.Position-myPos).Magnitude
if d<=150 and(d<=70 or pollFrame%4==0)then
local animator=animatorCache[model]
if not(animator and animator.Parent)then
animator=model:FindFirstChildWhichIsA("Animator",true)
animatorCache[model]=animator
end
if animator then
local ok,tracks=pcall(animator.GetPlayingAnimationTracks,animator)
if ok then
for i=1,#tracks do
markAndHandle(model,tracks[i],true)
end
end
end
end
end
end
end
end
end))

local inputIds={}
local syncing=false
local function addSlider(sec,key,title,desc,min,max,rounding,onChange)
rounding=rounding or 2
local sliderId="AB_"..key
local inputId=sliderId.."_Input"
local fmt="%."..rounding.."f"
table.insert(inputIds,inputId)
sec:AddSlider(sliderId,{
Title=title,Description=desc,
Min=min,Max=max,Default=CONFIG[key],Rounding=rounding,
Callback=function(v)
CONFIG[key]=v
if onChange then onChange(v)end
local input=Fluent.Options[inputId]
if input and not syncing then
syncing=true
pcall(function()input:SetValue(string.format(fmt,v))end)
syncing=false
end
end,
})
sec:AddInput(inputId,{
Title=title.."",
Default=string.format(fmt,CONFIG[key]),
Placeholder=min.." - "..max,
Numeric=false,
Finished=true,
Callback=function(text)
if syncing then return end
local n=tonumber(text)
syncing=true
if n then
n=math.clamp(n,min,max)
CONFIG[key]=n
if onChange then onChange(n)end
pcall(function()Fluent.Options[sliderId]:SetValue(n)end)
end
pcall(function()Fluent.Options[inputId]:SetValue(string.format(fmt,CONFIG[key]))end)
syncing=false
end,
})
end

local function unload()
alive=false
CONFIG.Enabled=false
pcall(clearAll)
for _,c in ipairs(connections)do
pcall(function()c:Disconnect()end)
end
pcall(stopCamlock)
pcall(destroyHighlight)
pcall(function()RunService:UnbindFromRenderStep(CAM_BIND)end)
pcall(function()Fluent:Destroy()end)
if Env.AutoBlockUnload==unload then
Env.AutoBlockUnload=nil
end
end
Env.AutoBlockUnload=unload

Description="Release F, destroy UI and stop everything",

})
local function loadAddon(name,urls)
if type(Fluent[name])=="table"then return Fluent[name]end
for _,url in ipairs(urls)do
local ok,result=pcall(function()return loadstring(game:HttpGet(url))()end)
if ok and type(result)=="table"and result.SetLibrary then return result end
end
return nil
end
